٢٩ شوال ١٣٦٩هـ (١٣ أغسطس ١٩٥٠م) · ورقة من المحكمة الشرعية ووثائق ملكية، كما نقلها الجزء السابع من المعالم

يوسف وعبدالوهاب ولدا عيسى يتخارجان عن كل ما بينهما

يوسف عيسى القطاميعبدالوهاب عيسى القطاميعيسى عبدالوهاب القطامي

ما هي

بموجب ورقة صادرة من المحكمة الشرعية بتاريخ ١٩٥٠/٨/٨م تخارج يوسف وعبدالوهاب ولدا عيسى القطامي عن جميع ما كان مشتركا بينهما من عقار وأثاث ودين وعروض تجارة، على أن يدفع عبدالوهاب لأخيه يوسف عشرين ألف روبية، وترك له يوسف أربعة بيوت. ثم سُجلت البيوت في فريج ابن رومي بالوثائق ١٢٢٥ و١٢٢٦ و١٢٢٧، ومنها بيتان يصل بينهما جسر، أحدهما اشتُري سنة ١٩٢٦ من عبداللطيف بن إبراهيم الدهيم والآخر سنة ١٩٣٧ من أبناء محمد بن علي الشيرازي، وبيت اشتراه الأخوان سنة ١٩٤٤ من رقية بنت أحمد الفضالة.

أهميتها

هي اللحظة التي اقتسم فيها ولدا عيسى تجارة العائلة وأملاكها بعد وفاة أبيهما بأكثر من عشرين سنة، وتبين حجم ما بنياه معا.

من المقصود

يوسف وعبدالوهاب ولدا عيسى القطامي كلاهما في شجرتنا، والورقة نفسها من المحكمة مذكورة في كل هذه البيوت.

صور الصفحات

النص الأصلي

الوثيقتان رقم ١٢٢٥ و١٢٢٦ جلد ١٤ المؤرختان ٢٩ شوال ١٣٦٩هـ (١٩٥٠/٨/١٣م): «ثبت بموجب الورقة الصادرة من المحكمة الشرعية رقم ٥٤ بتاريخ ١٩٥٠/٨/٨م أن يوسف وعبدالوهاب ولدي عيسى القطامي قد تخارجا عن جميع ما هو مشترك بينهما من عقار وأثاث ودين وعروض تجارة، وذلك نظير أن يدفع عبدالوهاب لأخيه يوسف مبلغ ٢٠ ألف روبية، وترك له أربعة بيوت منها هذين البيتين». حيث تملك يوسف وعبدالوهاب البيت الأول (الشمالي القبلي) بالشراء من عبداللطيف بن إبراهيم الدهيم بالوثيقة رقم ٩٤١ المؤرخة ٢٥ ربيع الأول ١٣٤٥هـ (١٩٢٦/١٠/٢١م)، وتملكا البيت الثاني (الجنوبي الشرقي) بالشراء من أحمد وعلي وعبدالله أبناء محمد بن علي ووالدتهم شريفة بنت محمد الإبراهيم بموجب الوثيقة رقم ٢٩٥ في ١٥ رمضان ١٣٥٦هـ (١٩٣٧/١١/١٩م). الوثيقة رقم ٤٧٦ المؤرخة ٥ شعبان ١٣٦٣هـ (١٩٤٤/٧/٢٦م): «...وقد شهد السيد أحمد بن السيد عقيل وعبدالله بن علي الغريب أن رقية بنت أحمد الفضالة باعت البيت على يوسف وعبدالوهاب ولدي عيسى القطامي».

ملاحظات

معالم مدينة الكويت القديمة، الجزء السابع، ص ١٠٣ و١٠٤ (القسيمتان ١٠١ و١٠٢)، وص ١١٢ (القسيمة ١٣١، الوثيقة ١٢٢٧). ينقل الكتاب من سجلات إدارة التسجيل، ولم نطلع على أصول الوثائق نفسها.
